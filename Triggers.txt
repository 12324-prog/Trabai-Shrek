-- TRIGGERS PARA INGREDIENTES
DROP TRIGGER IF EXISTS trg_insert_ingredientes;
DELIMITER $$
CREATE TRIGGER trg_insert_ingredientes
AFTER INSERT ON ingredientes
FOR EACH ROW
BEGIN
    -- associa o novo ingrediente ao último prato (se existir)
    INSERT INTO composicao (cod_prato, cod_ingrediente)
    SELECT MAX(cod_prato), NEW.cod_ingrediente FROM pratos;

    -- recalcula o valor_unitario dos pratos afetados
    UPDATE pratos p
    JOIN composicao c ON p.cod_prato = c.cod_prato
    SET p.valor_unitario = IFNULL(
        (SELECT SUM(i2.valor_unitario)
         FROM composicao c2
         JOIN ingredientes i2 ON c2.cod_ingrediente = i2.cod_ingrediente
         WHERE c2.cod_prato = p.cod_prato), 0
    ) + p.taxa_prato
    WHERE p.cod_prato IS NOT NULL;
END $$
DELIMITER ;

DROP TRIGGER IF EXISTS trg_update_ingredientes;
DELIMITER $$
CREATE TRIGGER trg_update_ingredientes
AFTER UPDATE ON ingredientes
FOR EACH ROW
BEGIN
    IF (NEW.valor_unitario <> OLD.valor_unitario) THEN
        UPDATE pratos p
        JOIN composicao c ON p.cod_prato = c.cod_prato
        SET p.valor_unitario = IFNULL(
            (SELECT SUM(i2.valor_unitario)
             FROM composicao c2
             JOIN ingredientes i2 ON c2.cod_ingrediente = i2.cod_ingrediente
             WHERE c2.cod_prato = p.cod_prato), 0
        ) + p.taxa_prato
        WHERE p.cod_prato IN (
            SELECT cod_prato FROM composicao WHERE cod_ingrediente = NEW.cod_ingrediente
        );
    END IF;
END $$
DELIMITER ;

DROP TRIGGER IF EXISTS trg_delete_ingredientes;
DELIMITER $$
CREATE TRIGGER trg_delete_ingredientes
BEFORE DELETE ON ingredientes
FOR EACH ROW
BEGIN
    -- Recalcula valor dos pratos que usam o ingrediente que será removido
    UPDATE pratos p
    JOIN composicao c ON p.cod_prato = c.cod_prato
    SET p.valor_unitario = IFNULL(
        (SELECT SUM(i2.valor_unitario)
         FROM composicao c2
         JOIN ingredientes i2 ON c2.cod_ingrediente = i2.cod_ingrediente
         WHERE c2.cod_prato = p.cod_prato), 0
    ) + p.taxa_prato
    WHERE p.cod_prato IN (
        SELECT cod_prato FROM composicao WHERE cod_ingrediente = OLD.cod_ingrediente
    );

    -- remove composições e itens de compra relacionados ao ingrediente que será removido
    DELETE FROM composicao WHERE cod_ingrediente = OLD.cod_ingrediente;
    DELETE FROM itens_compra WHERE cod_ingrediente = OLD.cod_ingrediente;
END $$
DELIMITER ;

-- TRIGGERS PARA ITENS_COMPRA
DROP TRIGGER IF EXISTS trg_insert_itens_compra;
DELIMITER $$
CREATE TRIGGER trg_insert_itens_compra
AFTER INSERT ON itens_compra
FOR EACH ROW
BEGIN
    UPDATE ingredientes
    SET quantidade_estoque = quantidade_estoque + NEW.quantidade
    WHERE cod_ingrediente = NEW.cod_ingrediente;

    UPDATE compras
    SET valor_total = IFNULL((
        SELECT SUM(quantidade * valor_unitario)
        FROM itens_compra
        WHERE cod_compra = NEW.cod_compra
    ), 0)
    WHERE cod_compra = NEW.cod_compra;
END $$
DELIMITER ;

DROP TRIGGER IF EXISTS trg_update_itens_compra;
DELIMITER $$
CREATE TRIGGER trg_update_itens_compra
AFTER UPDATE ON itens_compra
FOR EACH ROW
BEGIN
    UPDATE ingredientes
    SET quantidade_estoque = quantidade_estoque + (NEW.quantidade - OLD.quantidade)
    WHERE cod_ingrediente = NEW.cod_ingrediente;

    UPDATE compras
    SET valor_total = IFNULL((
        SELECT SUM(quantidade * valor_unitario)
        FROM itens_compra
        WHERE cod_compra = NEW.cod_compra
    ), 0)
    WHERE cod_compra = NEW.cod_compra;
END $$
DELIMITER ;

DROP TRIGGER IF EXISTS trg_delete_itens_compra;
DELIMITER $$
CREATE TRIGGER trg_delete_itens_compra
AFTER DELETE ON itens_compra
FOR EACH ROW
BEGIN
    UPDATE ingredientes
    SET quantidade_estoque = quantidade_estoque - OLD.quantidade
    WHERE cod_ingrediente = OLD.cod_ingrediente;

    UPDATE compras
    SET valor_total = IFNULL((
        SELECT SUM(quantidade * valor_unitario)
        FROM itens_compra
        WHERE cod_compra = OLD.cod_compra
    ), 0)
    WHERE cod_compra = OLD.cod_compra;
END $$
DELIMITER ;

-- TRIGGERS PARA ITENS_PEDIDO
DROP TRIGGER IF EXISTS trg_insert_itens_pedido;
DELIMITER $$
CREATE TRIGGER trg_insert_itens_pedido
AFTER INSERT ON itens_pedido
FOR EACH ROW
BEGIN
    UPDATE pedidos
    SET valor_pago = IFNULL(valor_pago, 0) + (NEW.valor_unitario * NEW.quantidade)
    WHERE cod_pedido = NEW.cod_pedido;
END $$
DELIMITER ;

DROP TRIGGER IF EXISTS trg_update_itens_pedido;
DELIMITER $$
CREATE TRIGGER trg_update_itens_pedido
AFTER UPDATE ON itens_pedido
FOR EACH ROW
BEGIN
    IF (NEW.cod_pedido <> OLD.cod_pedido) THEN
        -- item movido entre pedidos
        UPDATE pedidos
        SET valor_pago = valor_pago - (OLD.valor_unitario * OLD.quantidade)
        WHERE cod_pedido = OLD.cod_pedido;

        UPDATE pedidos
        SET valor_pago = IFNULL(valor_pago, 0) + (NEW.valor_unitario * NEW.quantidade)
        WHERE cod_pedido = NEW.cod_pedido;
    ELSEIF (OLD.quantidade <> NEW.quantidade) OR (OLD.valor_unitario <> NEW.valor_unitario) THEN
        -- item alterado no mesmo pedido
        UPDATE pedidos
        SET valor_pago = IFNULL(valor_pago, 0) + (NEW.valor_unitario * NEW.quantidade) - (OLD.valor_unitario * OLD.quantidade)
        WHERE cod_pedido = NEW.cod_pedido;
    END IF;
END $$
DELIMITER ;

DROP TRIGGER IF EXISTS trg_delete_itens_pedido;
DELIMITER $$
CREATE TRIGGER trg_delete_itens_pedido
AFTER DELETE ON itens_pedido
FOR EACH ROW
BEGIN
    UPDATE pedidos
    SET valor_pago = valor_pago - (OLD.valor_unitario * OLD.quantidade)
    WHERE cod_pedido = OLD.cod_pedido;
END $$
DELIMITER ;

-- TRIGGER PARA PEDIDOS (Ao deletar um pedido, remover seus itens)
DROP TRIGGER IF EXISTS trg_delete_pedido;
DELIMITER $$
CREATE TRIGGER trg_delete_pedido
BEFORE DELETE ON pedidos
FOR EACH ROW
BEGIN
    DELETE FROM itens_pedido WHERE cod_pedido = OLD.cod_pedido;
END $$
DELIMITER ;

-- TRIGGERS PARA PRATOS
DROP TRIGGER IF EXISTS trg_update_pratos;
DELIMITER $$
CREATE TRIGGER trg_update_pratos
AFTER UPDATE ON pratos
FOR EACH ROW
BEGIN
    IF (OLD.taxa_prato <> NEW.taxa_prato) THEN
        UPDATE pratos
        SET valor_unitario = valor_unitario - OLD.taxa_prato + NEW.taxa_prato
        WHERE cod_prato = OLD.cod_prato;
    END IF;

    -- atualiza preço nos itens_pedido quando preço do prato muda
    UPDATE itens_pedido
    SET valor_unitario = NEW.valor_unitario
    WHERE cod_prato = OLD.cod_prato;
END $$
DELIMITER ;

DROP TRIGGER IF EXISTS trg_delete_pratos;
DELIMITER $$
CREATE TRIGGER trg_delete_pratos
AFTER DELETE ON pratos
FOR EACH ROW
BEGIN
    -- ajusta valor_pago dos pedidos que tinham itens desse prato
    UPDATE pedidos p
    JOIN itens_pedido ip ON p.cod_pedido = ip.cod_pedido
    SET p.valor_pago = p.valor_pago - (ip.valor_unitario * ip.quantidade)
    WHERE ip.cod_prato = OLD.cod_prato;

    -- remove itens_pedido relacionados ao prato
    DELETE FROM itens_pedido WHERE cod_prato = OLD.cod_prato;

    -- remove composições do prato
    DELETE FROM composicao WHERE cod_prato = OLD.cod_prato;
END $$
DELIMITER ;
