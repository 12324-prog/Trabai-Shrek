-- ==============================================
-- FUNÇÃO PARA RECALCULAR VALOR DO PRATO
-- ==============================================
DROP FUNCTION IF EXISTS calcular_valor_prato;
DELIMITER $$
CREATE FUNCTION calcular_valor_prato(prato_id INT)
RETURNS DECIMAL(10,2)
BEGIN
    RETURN (
        SELECT IFNULL(SUM(i.valor_unitario),0)
        FROM composicao c
        JOIN ingredientes i ON c.cod_ingrediente = i.cod_ingrediente
        WHERE c.cod_prato = prato_id
    );
END$$
DELIMITER ;

-- TRIGGERS PARA COMPOSICAO
DROP TRIGGER IF EXISTS trg_insert_composicao;
DELIMITER $$
CREATE TRIGGER trg_insert_composicao
AFTER INSERT ON composicao
FOR EACH ROW
BEGIN

    UPDATE pratos p
    JOIN composicao c ON p.cod_prato = c.cod_prato
    SET p.valor_unitario = calcular_valor_prato(p.cod_prato) + p.taxa_prato
    WHERE c.cod_ingrediente = NEW.cod_ingrediente;
END$$
DELIMITER ;

DROP TRIGGER IF EXISTS trg_update_ingredientes;
DELIMITER $$
CREATE TRIGGER trg_update_ingredientes
AFTER UPDATE ON ingredientes
FOR EACH ROW
BEGIN
    IF (OLD.valor_unitario <> NEW.valor_unitario) THEN
        UPDATE pratos p
        JOIN composicao c ON p.cod_prato = c.cod_prato
        SET p.valor_unitario = calcular_valor_prato(p.cod_prato) + p.taxa_prato
        WHERE c.cod_ingrediente = NEW.cod_ingrediente;
    END IF;
END$$
DELIMITER ;

-- AJUSTAR DELETE
DROP TRIGGER IF EXISTS trg_delete_ingredientes;
DELIMITER $$
CREATE TRIGGER trg_delete_ingredientes
BEFORE DELETE ON ingredientes
FOR EACH ROW
BEGIN
    UPDATE pratos p
    JOIN composicao c ON p.cod_prato = c.cod_prato
    SET p.valor_unitario = 0
    WHERE c.cod_ingrediente = OLD.cod_ingrediente;

    DELETE FROM composicao WHERE cod_ingrediente = OLD.cod_ingrediente;
    SET @deletando_ingrediente = TRUE;
    DELETE FROM itens_compra WHERE cod_ingrediente = OLD.cod_ingrediente;
    SET @deletando_ingrediente = FALSE;
END$$
DELIMITER ;

-- TRIGGERS PARA ITENS_COMPRA
DROP TRIGGER IF EXISTS trg_insert_itens_compra;
DELIMITER $$
CREATE TRIGGER trg_insert_itens_compra
AFTER INSERT ON itens_compra
FOR EACH ROW
BEGIN
    UPDATE ingredientes
    SET quantidade_estoque = quantidade_estoque + NEW.quantidade
    WHERE cod_ingrediente = NEW.cod_ingrediente;

    UPDATE compras
    SET valor_total = (
        SELECT IFNULL(SUM(quantidade * valor_unitario), 0)
        FROM itens_compra
        WHERE cod_compra = NEW.cod_compra
    )
    WHERE cod_compra = NEW.cod_compra;
END$$
DELIMITER ;

DROP TRIGGER IF EXISTS trg_update_itens_compra;
DELIMITER $$
CREATE TRIGGER trg_update_itens_compra
AFTER UPDATE ON itens_compra
FOR EACH ROW
BEGIN
    UPDATE ingredientes
    SET quantidade_estoque = quantidade_estoque + (NEW.quantidade - OLD.quantidade)
    WHERE cod_ingrediente = NEW.cod_ingrediente;

    UPDATE compras
    SET valor_total = (
        SELECT IFNULL(SUM(quantidade * valor_unitario), 0)
        FROM itens_compra
        WHERE cod_compra = NEW.cod_compra
    )
    WHERE cod_compra = NEW.cod_compra;
END$$
DELIMITER ;

DROP TRIGGER IF EXISTS trg_delete_itens_compra;
DELIMITER $$
CREATE TRIGGER trg_delete_itens_compra
AFTER DELETE ON itens_compra
FOR EACH ROW
BEGIN
	IF (@deletando_ingrediente IS NULL OR @deletando_ingrediente = FALSE) THEN
        UPDATE ingredientes
        SET quantidade_estoque = quantidade_estoque - OLD.quantidade
        WHERE cod_ingrediente = OLD.cod_ingrediente;
    END IF;

    UPDATE compras
    SET valor_total = (
        SELECT IFNULL(SUM(quantidade * valor_unitario), 0)
        FROM itens_compra
        WHERE cod_compra = OLD.cod_compra
    )
    WHERE cod_compra = OLD.cod_compra;
END$$
DELIMITER ;

-- TRIGGERS PARA ITENS_PEDIDO
DROP TRIGGER IF EXISTS trg_insert_itens_pedido;
DELIMITER $$
CREATE TRIGGER trg_insert_itens_pedido
AFTER INSERT ON itens_pedido
FOR EACH ROW
BEGIN
    UPDATE pedidos
    SET valor_pago = (
        SELECT IFNULL(SUM(quantidade * valor_unitario),0)
        FROM itens_pedido
        WHERE cod_pedido = NEW.cod_pedido
    )
    WHERE cod_pedido = NEW.cod_pedido;
    
    UPDATE ingredientes as i JOIN composicao as c 
    ON (i.cod_ingrediente = c.cod_ingrediente)
    SET quantidade_estoque = (quantidade_estoque - NEW.quantidade)
    WHERE cod_prato = NEW.cod_prato;
END$$
DELIMITER ;

DROP TRIGGER IF EXISTS trg_update_itens_pedido;
DELIMITER $$
CREATE TRIGGER trg_update_itens_pedido
AFTER UPDATE ON itens_pedido
FOR EACH ROW
BEGIN
    UPDATE pedidos
    SET valor_pago = (
        SELECT IFNULL(SUM(quantidade * valor_unitario),0)
        FROM itens_pedido
        WHERE cod_pedido = NEW.cod_pedido
    )
    WHERE cod_pedido = NEW.cod_pedido;

    IF (NEW.cod_pedido <> OLD.cod_pedido) THEN
        UPDATE pedidos
        SET valor_pago = (
            SELECT IFNULL(SUM(quantidade * valor_unitario),0)
            FROM itens_pedido
            WHERE cod_pedido = OLD.cod_pedido
        )
        WHERE cod_pedido = OLD.cod_pedido;
    END IF;
    IF (NEW.quantidade <> OLD.quantidade) THEN
        UPDATE ingredientes as i JOIN composicao as c 
        ON (i.cod_ingrediente = c.cod_ingrediente)
        SET quantidade_estoque = (quantidade_estoque - NEW.quantidade)
        WHERE cod_prato = OLD.cod_prato;
    END IF;
END$$
DELIMITER ;

DROP TRIGGER IF EXISTS trg_delete_itens_pedido;
DELIMITER $$
CREATE TRIGGER trg_delete_itens_pedido
AFTER DELETE ON itens_pedido
FOR EACH ROW
BEGIN
    IF (@deletando_pedido IS NULL OR @deletando_pedido = FALSE) THEN
        UPDATE pedidos
        SET valor_pago = (
            SELECT IFNULL(SUM(quantidade * valor_unitario), 0)
            FROM itens_pedido
            WHERE cod_pedido = OLD.cod_pedido
        )
        WHERE cod_pedido = OLD.cod_pedido;
    END IF;
    
    UPDATE ingredientes as i JOIN composicao as c 
    ON (i.cod_ingrediente = c.cod_ingrediente)
    SET quantidade_estoque = (quantidade_estoque + OLD.quantidade)
    WHERE cod_prato = OLD.cod_prato;
END$$
DELIMITER ;

-- TRIGGER PARA PEDIDOS
DROP TRIGGER IF EXISTS trg_delete_pedido;
DELIMITER $$
CREATE TRIGGER trg_delete_pedido
BEFORE DELETE ON pedidos
FOR EACH ROW
BEGIN
    SET @deletando_pedido = TRUE;
    DELETE FROM itens_pedido WHERE cod_pedido = OLD.cod_pedido;
    SET @deletando_pedido = NULL;
END $$
DELIMITER ;

-- TRIGGERS PARA PRATOS
DROP TRIGGER IF EXISTS trg_update_pratos;
DELIMITER $$
CREATE TRIGGER trg_update_pratos
BEFORE UPDATE ON pratos
FOR EACH ROW
BEGIN
    IF (OLD.taxa_prato <> NEW.taxa_prato) THEN
        SET NEW.valor_unitario = calcular_valor_prato(NEW.cod_prato) + NEW.taxa_prato;
    END IF;
END$$
DELIMITER ;

DROP TRIGGER IF EXISTS trg_after_update_pratos;
DELIMITER $$
CREATE TRIGGER trg_after_update_pratos
AFTER UPDATE ON pratos
FOR EACH ROW
BEGIN
    IF (OLD.valor_unitario <> NEW.valor_unitario) THEN
        UPDATE itens_pedido
        SET valor_unitario = NEW.valor_unitario
        WHERE cod_prato = NEW.cod_prato;
    END IF;
END$$
DELIMITER ;


DROP TRIGGER IF EXISTS trg_delete_pratos;
DELIMITER $$
CREATE TRIGGER trg_delete_pratos
BEFORE DELETE ON pratos
FOR EACH ROW
BEGIN
    UPDATE pedidos p
    JOIN itens_pedido ip ON p.cod_pedido = ip.cod_pedido
    SET p.valor_pago = p.valor_pago - (ip.valor_unitario * ip.quantidade)
    WHERE ip.cod_prato = OLD.cod_prato;

    DELETE FROM itens_pedido WHERE cod_prato = OLD.cod_prato;

    DELETE FROM composicao WHERE cod_prato = OLD.cod_prato;
END $$
DELIMITER ;

DROP TRIGGER IF EXISTS cliente_delete;
DELIMITER $$
CREATE TRIGGER cliente_delete
BEFORE DELETE ON clientes
FOR EACH ROW
BEGIN
    DELETE FROM itens_pedido
    WHERE cod_pedido IN (SELECT cod_pedido FROM pedidos WHERE cod_cliente = OLD.cod_cliente);

    DELETE FROM pedidos
    WHERE cod_cliente = OLD.cod_cliente;
END$$
DELIMITER ;
